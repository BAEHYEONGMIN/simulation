# ============================================================
# 6. 기본 importance
# ============================================================
DEFAULT_IMPORTANCE_BY_TYPE: Dict[str, int] = {
    "profile": 9,
    "style": 8,
    "preference": 6,
    "taboo": 9,
    "relationship": 7,
    "fact": 5,
}


# ============================================================
# 7. 기본 confidence
# ============================================================
# 추출 방식에 따라 초기 confidence를 다르게 둘 수 있다.
# ============================================================
DEFAULT_CONFIDENCE_BY_SOURCE: Dict[str, float] = {
    "explicit": 0.90,     # 사용자가 직접 명시적으로 말함
    "repeated": 0.95,     # 반복 확인됨
    "inferred": 0.65,     # 문맥상 추론
    "weak": 0.45,         # 애매함
}


# ============================================================
# 8. 메모리 후보 데이터 모델
# ============================================================
@dataclass
class MemoryCandidate:
    memory_type: str
    memory_key: str
    memory_value: str
    importance: Optional[int] = None
    confidence: Optional[float] = None


# ============================================================
# 9. 유틸 함수
# ============================================================
def is_allowed_memory_type(memory_type: str) -> bool:
    return memory_type in ALLOWED_MEMORY_TYPES


def is_allowed_memory_key(memory_type: str, memory_key: str) -> bool:
    return memory_key in ALLOWED_MEMORY_KEYS.get(memory_type, [])


def normalize_memory_value(memory_key: str, raw_value: str) -> str:
    """
    raw_value를 가능한 경우 canonical value로 변환한다.
    변환 실패 시 원문을 trim해서 반환한다.
    """
    value = raw_value.strip()

    # exact lower match
    lower_value = value.lower()

    # 1) normalization map 우선
    key_map = NORMALIZATION_MAP.get(memory_key, {})
    for raw, normalized in key_map.items():
        if lower_value == raw.lower():
            return normalized

    # 2) 이미 canonical value면 그대로 통과
    allowed_values = NORMALIZED_VALUE_OPTIONS.get(memory_key, [])
    for option in allowed_values:
        if lower_value == option.lower():
            return option

    # 3) 못 맞추면 원문 반환
    return value


def get_default_importance(memory_type: str) -> int:
    return DEFAULT_IMPORTANCE_BY_TYPE.get(memory_type, 5)


def get_default_confidence(source_kind: str) -> float:
    return DEFAULT_CONFIDENCE_BY_SOURCE.get(source_kind, 0.70)


def validate_candidate(candidate: MemoryCandidate) -> List[str]:
    """
    LLM이 뽑은 메모리 후보가 최소한 스키마에 맞는지 검사한다.
    반환값: 에러 메시지 목록. 비어 있으면 통과.
    """
    errors: List[str] = []

    if not is_allowed_memory_type(candidate.memory_type):
        errors.append(f"invalid memory_type: {candidate.memory_type}")

    if not is_allowed_memory_key(candidate.memory_type, candidate.memory_key):
        errors.append(
            f"invalid memory_key for type {candidate.memory_type}: {candidate.memory_key}"
        )

    if not candidate.memory_value or not candidate.memory_value.strip():
        errors.append("memory_value is empty")

    return errors


def build_memory_record(
    *,
    conf_uid: str,
    owner_speaker_id: str,
    target_speaker_id: str,
    source_history_uid: Optional[str],
    source_message_id: Optional[int],
    candidate: MemoryCandidate,
    source_kind: str = "explicit",
) -> Dict[str, Any]:
    """
    user_memories insert/upsert용 dict를 만든다.
    """
    normalized_value = normalize_memory_value(candidate.memory_key, candidate.memory_value)

    importance = candidate.importance
    if importance is None:
        importance = get_default_importance(candidate.memory_type)

    confidence = candidate.confidence
    if confidence is None:
        confidence = get_default_confidence(source_kind)

    return {
        "conf_uid": conf_uid,
        "owner_speaker_id": owner_speaker_id,
        "target_speaker_id": target_speaker_id,
        "memory_type": candidate.memory_type,
        "memory_key": candidate.memory_key,
        "memory_value": normalized_value,
        "source_history_uid": source_history_uid,
        "source_message_id": source_message_id,
        "importance": importance,
        "confidence": confidence,
        "evidence_count": 1,
        "status": "active",
    }

# 예시 프롬프트:
#
# 너의 작업은 대화에서 장기 기억 후보를 추출하는 것이다.
#
# 허용된 memory_type:
# - profile
# - style
# - preference
# - taboo
# - relationship
# - fact
#
# 허용된 memory_key:
# - profile.user_name
# - style.preferred_speech_style
# - style.preferred_response_length
# - preference.favorite_genre
# - preference.favorite_topic
# - taboo.disliked_topic
# - taboo.avoid_behavior
# - relationship.attitude_to_character
# - relationship.closeness_level
# - fact.current_activity
# - fact.current_goal
#
# value 정규화 규칙:
# - preferred_speech_style은 polite / casual 중 하나
# - preferred_response_length는 short / medium / long 중 하나
# - favorite_genre는 fantasy / sci_fi / romance / mystery / horror / slice_of_life 중 하나
# - favorite_topic은 technology / games / anime / novels / movies 중 하나
# - disliked_topic은 politics / religion / personal_life 중 하나
# - avoid_behavior는 rude_tone / commanding_tone / personal_questions 중 하나
# - attitude_to_character는 respectful / playful / commanding / distant / friendly 중 하나
# - closeness_level은 low / medium / high 중 하나
#
# 저장 가치가 있는 정보만 JSON 배열로 출력하라.
# 형식:
# [
#   {
#     "memory_type": "...",
#     "memory_key": "...",
#     "memory_value": "...",
#     "confidence": 0.0
#   }
# ]
#
# 모르면 추출하지 마라.
# 설명 문장은 쓰지 마라.
#
# ------------------------------------------------------------
# [백엔드 처리 흐름 예시]
# ------------------------------------------------------------
#
# 1. chat_messages에서 최근 3~10턴 또는 이번 턴 블록을 읽는다.
# 2. 위 프롬프트로 LLM에게 memory 후보를 구조화 추출시킨다.
# 3. JSON을 파싱한다.
# 4. validate_candidate()로 type/key/value 최소 검증한다.
# 5. normalize_memory_value()로 canonical value로 정규화한다.
# 6. user_memories에서 같은
#    (conf_uid, owner_speaker_id, target_speaker_id, memory_type, memory_key)
#    active row가 있는지 찾는다.
# 7. 있으면:
#    - memory_value 갱신 여부 판단
#    - confidence 증가 또는 충돌 처리
#    - evidence_count += 1
#    - last_seen_at 갱신
# 8. 없으면 insert 한다.
#
# ------------------------------------------------------------
# [중요한 운영 원칙]
# ------------------------------------------------------------
#
# 1) LLM 출력값을 그대로 DB에 넣지 말 것
#    - 반드시 validate + normalize를 거쳐라.
#
# 2) memory_type, memory_key는 백엔드가 강하게 통제할 것
#    - 이 파일이 source of truth다.
#
# 3) memory_value는 일부만 강하게 정규화할 것
#    - 검색/필터링할 값만 canonical value로 통일해라.
#
# 4) 장기 기억 추출은 응답 전이 아니라 응답 후 비동기 처리 권장
#    - 현재 응답 지연을 줄일 수 있다.
#
# ------------------------------------------------------------
# [DB 컬럼과 이 파일의 연결]
# ------------------------------------------------------------
#
# chatbot.user_memories
# - memory_type   <- candidate.memory_type
# - memory_key    <- candidate.memory_key
# - memory_value  <- normalize_memory_value() 결과
# - importance    <- get_default_importance()
# - confidence    <- get_default_confidence() 또는 LLM 출력
#
# chatbot.chat_messages
# - source_history_uid
# - source_message_id
#   는 user_memories에 출처로 기록 가능
#
# chatbot.conversation_summaries
# - 요약문은 장기 기억 추출과 별도다.
# - 같은 턴 블록을 기반으로 summary와 memory extraction을 각각 돌릴 수 있다.
#
# chatbot.documents
