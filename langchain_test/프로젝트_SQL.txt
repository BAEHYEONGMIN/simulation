-- =========================================================
-- 0. schema / extension
-- =========================================================
create schema if not exists chatbot;

create extension if not exists vector;

-- =========================================================
-- 1. updated_at trigger function
-- =========================================================
create or replace function chatbot.set_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

-- =========================================================
-- 2. chat_messages
-- 원본 대화 로그 (다중 화자 대응)
-- =========================================================
create table if not exists chatbot.chat_messages (
  id bigserial not null,

  conf_uid text not null,
  history_uid text not null,

  -- 메시지 역할
  role text not null,

  -- 실제 발화자 구분
  speaker_type text not null,
  speaker_id text not null,
  display_name text null,

  content text not null,
  avatar text null,

  created_at timestamptz not null default now(),

  constraint chat_messages_pkey primary key (id),

  constraint chat_messages_role_check
    check (role in ('system', 'human', 'ai', 'tool')),

  constraint chat_messages_speaker_type_check
    check (speaker_type in ('user', 'character', 'system', 'tool'))
);

create index if not exists idx_chat_messages_history
  on chatbot.chat_messages (conf_uid, history_uid);

create index if not exists idx_chat_messages_history_created
  on chatbot.chat_messages (conf_uid, history_uid, created_at desc);

create index if not exists idx_chat_messages_speaker
  on chatbot.chat_messages (speaker_id, created_at desc);

create index if not exists idx_chat_messages_created_at
  on chatbot.chat_messages (created_at desc);

-- =========================================================
-- 3. conversation_summaries
-- 세션 요약 저장
-- =========================================================
create table if not exists chatbot.conversation_summaries (
  id bigserial not null,

  conf_uid text not null,
  history_uid text not null,

  summary_type text not null default 'rolling',
  summary_text text not null,

  start_message_id bigint null,
  end_message_id bigint null,

  covered_message_count integer not null default 0,
  summary_seq integer not null default 1,

  -- 참여자 목록
  participants jsonb not null default '[]'::jsonb,

  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),

  constraint conversation_summaries_pkey primary key (id),

  constraint conversation_summaries_summary_type_check
    check (summary_type in ('rolling', 'checkpoint', 'final'))
);

create index if not exists idx_conversation_summaries_session
  on chatbot.conversation_summaries (conf_uid, history_uid, summary_seq desc);

create index if not exists idx_conversation_summaries_range
  on chatbot.conversation_summaries (conf_uid, history_uid, end_message_id desc);

create index if not exists idx_conversation_summaries_created
  on chatbot.conversation_summaries (created_at desc);

create index if not exists idx_conversation_summaries_participants_gin
  on chatbot.conversation_summaries
  using gin (participants);

drop trigger if exists trg_conversation_summaries_updated_at on chatbot.conversation_summaries;
create trigger trg_conversation_summaries_updated_at
before update on chatbot.conversation_summaries
for each row
execute function chatbot.set_updated_at();

-- =========================================================
-- 4. user_memories
-- 장기 기억 저장
-- 다중 화자 대응:
-- owner_speaker_id  : 이 기억을 보유/사용하는 주체
-- target_speaker_id : 이 기억이 가리키는 대상 화자
-- =========================================================
create table if not exists chatbot.user_memories (
  id bigserial not null,

  conf_uid text not null,

  owner_speaker_id text not null,
  target_speaker_id text not null,

  memory_type text not null,
  memory_key text null,
  memory_value text not null,

  source_history_uid text null,
  source_message_id bigint null,

  importance integer not null default 5,
  confidence numeric(4,3) not null default 0.700,
  evidence_count integer not null default 1,

  status text not null default 'active',

  first_seen_at timestamptz not null default now(),
  last_seen_at timestamptz not null default now(),
  last_accessed_at timestamptz null,
  access_count integer not null default 0,

  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),

  constraint user_memories_pkey primary key (id),

  constraint user_memories_importance_check
    check (importance between 1 and 10),

  constraint user_memories_confidence_check
    check (confidence >= 0 and confidence <= 1),

  constraint user_memories_status_check
    check (status in ('active', 'archived', 'deleted', 'conflicted')),

  constraint user_memories_memory_type_check
    check (memory_type in ('profile', 'style', 'preference', 'taboo', 'relationship', 'fact'))
);

create index if not exists idx_user_memories_owner_target
  on chatbot.user_memories (conf_uid, owner_speaker_id, target_speaker_id);

create index if not exists idx_user_memories_lookup
  on chatbot.user_memories (owner_speaker_id, target_speaker_id, memory_type, importance desc);

create index if not exists idx_user_memories_active
  on chatbot.user_memories (owner_speaker_id, target_speaker_id, status, last_seen_at desc);

create index if not exists idx_user_memories_source
  on chatbot.user_memories (source_history_uid, source_message_id);

create unique index if not exists uq_user_memories_key
  on chatbot.user_memories (
    conf_uid,
    owner_speaker_id,
    target_speaker_id,
    memory_type,
    memory_key
  )
  where status = 'active' and memory_key is not null;

drop trigger if exists trg_user_memories_updated_at on chatbot.user_memories;
create trigger trg_user_memories_updated_at
before update on chatbot.user_memories
for each row
execute function chatbot.set_updated_at();

-- =========================================================
-- 5. documents
-- RAG / vector search용 문서
-- OpenAI text-embedding-3-small 기준 1536 차원
-- =========================================================
create table if not exists chatbot.documents (
  id bigserial not null,

  content text not null,
  metadata jsonb not null default '{}'::jsonb,
  embedding vector(1536) null,

  created_at timestamptz not null default now(),

  constraint documents_pkey primary key (id)
);

create index if not exists idx_documents_created_at
  on chatbot.documents (created_at desc);

create index if not exists idx_documents_metadata_gin
  on chatbot.documents
  using gin (metadata);

create index if not exists idx_documents_embedding
  on chatbot.documents
  using ivfflat (embedding vector_cosine_ops)
  with (lists = 100);

-- =========================================================
-- 6. vector search function
-- metadata filter 지원
-- =========================================================
create or replace function chatbot.match_documents (
  query_embedding vector(1536),
  match_count int default 5,
  filter jsonb default '{}'::jsonb
)
returns table (
  id bigint,
  content text,
  metadata jsonb,
  similarity float
)
language plpgsql
as $$
begin
  return query
  select
    d.id,
    d.content,
    d.metadata,
    1 - (d.embedding <=> query_embedding) as similarity
  from chatbot.documents d
  where d.metadata @> filter
  order by d.embedding <=> query_embedding
  limit match_count;
end;
$$;

-- =========================================================
-- documents_gemini 테이블 생성
-- 기존 documents(384차원) 구조와 동일, 768차원으로 변경
-- =========================================================
-- 1. 기존에 혹시 있다면 정리
drop function if exists chatbot.match_documents_gemini(vector(768), int, jsonb);
drop index if exists chatbot.idx_documents_gemini_embedding;
drop table if exists chatbot.documents_gemini;
-- 2. 테이블 생성 (768차원)
create table chatbot.documents_gemini (
  id bigserial not null,
  content text not null,
  metadata jsonb not null default '{}'::jsonb,
  embedding vector(768) null,
  created_at timestamptz not null default now(),
  constraint documents_gemini_pkey primary key (id)
);
-- 3. 인덱스 생성
create index if not exists idx_documents_gemini_created_at
  on chatbot.documents_gemini (created_at desc);
create index if not exists idx_documents_gemini_metadata_gin
  on chatbot.documents_gemini
  using gin (metadata);
create index if not exists idx_documents_gemini_embedding
  on chatbot.documents_gemini
  using ivfflat (embedding vector_cosine_ops)
  with (lists = 100);
-- 4. 벡터 검색 함수 생성
create or replace function chatbot.match_documents_gemini (
  query_embedding vector(768),
  match_count int default 5,
  filter jsonb default '{}'::jsonb
)
returns table (
  id bigint,
  content text,
  metadata jsonb,
  similarity float
)
language plpgsql
as $$
begin
  return query
  select
    d.id,
    d.content,
    d.metadata,
    1 - (d.embedding <=> query_embedding) as similarity
  from chatbot.documents_gemini d
  where d.metadata @> filter
  order by d.embedding <=> query_embedding
  limit match_count;
end;
$$;
